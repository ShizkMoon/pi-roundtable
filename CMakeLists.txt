cmake_minimum_required(VERSION 3.25)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PI_ROUNDTABLE_VERSION LIMIT_COUNT 1)
if(NOT PI_ROUNDTABLE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "VERSION must contain one three-part numeric product version without leading zeroes")
endif()
string(REPLACE "." ";" PI_ROUNDTABLE_VERSION_PARTS "${PI_ROUNDTABLE_VERSION}")
list(GET PI_ROUNDTABLE_VERSION_PARTS 0 PI_ROUNDTABLE_VERSION_MAJOR)
list(GET PI_ROUNDTABLE_VERSION_PARTS 1 PI_ROUNDTABLE_VERSION_MINOR)
list(GET PI_ROUNDTABLE_VERSION_PARTS 2 PI_ROUNDTABLE_VERSION_PATCH)
if(PI_ROUNDTABLE_VERSION_MAJOR GREATER 255 OR
   PI_ROUNDTABLE_VERSION_MINOR GREATER 255 OR
   PI_ROUNDTABLE_VERSION_PATCH GREATER 65535)
    message(FATAL_ERROR "VERSION exceeds Windows Installer limits (major/minor <= 255 and patch <= 65535)")
endif()

project(
    PiRoundtable
    VERSION ${PI_ROUNDTABLE_VERSION}
    DESCRIPTION "Deterministic meeting state core for Pi Roundtable"
    LANGUAGES CXX
)

option(PI_ROUNDTABLE_BUILD_TESTS "Build Pi Roundtable core tests" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(PI_ROUNDTABLE_BUILD_TESTS)
    include(CTest)
endif()

add_subdirectory(core)
